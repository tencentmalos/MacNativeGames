cmake_minimum_required(VERSION 3.22)
project(macnative_macos LANGUAGES C CXX OBJC OBJCXX ASM)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    message(FATAL_ERROR "The native Switch host requires Apple Silicon macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../android/dependencies.json" dependency_lock)
foreach(dependency foundation volk d2r_vulkan)
    if(dependency STREQUAL "foundation")
        set(source_dir "${FOUNDATION_SOURCE_DIR}")
    elseif(dependency STREQUAL "volk")
        set(source_dir "${VOLK_SOURCE_DIR}")
    else()
        set(source_dir "${D2R_SOURCE_DIR}")
    endif()
    string(JSON expected GET "${dependency_lock}" "${dependency}" commit)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${source_dir}" rev-parse HEAD
        OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
    if(NOT result EQUAL 0 OR NOT actual STREQUAL expected)
        message(FATAL_ERROR "${dependency}: checkout must match dependencies.json")
    endif()
endforeach()
if(NOT EXISTS "${VULKAN_HEADERS_DIR}/vulkan/vulkan.h")
    message(FATAL_ERROR "Set VULKAN_HEADERS_DIR to the managed Vulkan-Headers include directory")
endif()
set(VOLK_PULL_IN_VULKAN OFF)
set(VOLK_STATIC_DEFINES VK_USE_PLATFORM_METAL_EXT)
add_subdirectory("${VOLK_SOURCE_DIR}" volk)
target_include_directories(volk PUBLIC "${VULKAN_HEADERS_DIR}")
add_library(mng_vulkan_loader STATIC vulkan_driver.cpp)
target_include_directories(mng_vulkan_loader PUBLIC ../android/native)
target_link_libraries(mng_vulkan_loader PUBLIC volk)
target_compile_definitions(mng_vulkan_loader PUBLIC VK_NO_PROTOTYPES VK_ENABLE_BETA_EXTENSIONS VK_USE_PLATFORM_METAL_EXT)
target_compile_options(mng_vulkan_loader PUBLIC "$<$<COMPILE_LANGUAGE:C,CXX>:-include${VOLK_SOURCE_DIR}/volk.h>")
add_subdirectory(../porting porting)
add_subdirectory(../porting/profiler profiler)
add_subdirectory(../porting/nvn_vulkan nvn-vulkan)
if(MNG_BOTW_CONFIG_DIR)
    add_subdirectory(../porting/overlay overlay)
endif()
include(native_targets.cmake)
