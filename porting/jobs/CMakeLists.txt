# Import only Foundation's task infrastructure, with the same SDK ring owner.
include(../cmake/fmt.cmake)
add_library(foundation_host_fmt INTERFACE)
add_library(spatial::foundation_host_fmt ALIAS foundation_host_fmt)
target_link_libraries(foundation_host_fmt INTERFACE fmt::fmt)
add_library(mng_jobs_json INTERFACE)
add_library(spatial::third_party_nlohmann_json ALIAS mng_jobs_json)
target_include_directories(mng_jobs_json INTERFACE "${FOUNDATION_SOURCE_DIR}/third_party/nlohmann_json/include")
set(MI_OVERRIDE OFF CACHE BOOL "" FORCE)
set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(MI_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(MI_BUILD_OBJECT OFF CACHE BOOL "" FORCE)
set(MI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(ANDROID)
    set(MI_LOCAL_DYNAMIC_TLS ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${FOUNDATION_SOURCE_DIR}/third_party/mimalloc" mimalloc EXCLUDE_FROM_ALL)
if(ANDROID AND CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    target_compile_definitions(mimalloc-static PRIVATE MI_TLS_SLOT=2 MI_TLS_SLOT_FALLBACK=1)
endif()
add_library(spatial::third_party_mimalloc ALIAS mimalloc-static)
foreach(module modules/property basic/allocator basic/underlying/core basic/meta/reflection basic/async/async_task)
    string(REPLACE "/" "-" build_name "${module}")
    add_subdirectory("${FOUNDATION_SOURCE_DIR}/${module}" "${build_name}" EXCLUDE_FROM_ALL)
endforeach()
# Standalone profiler_ring already owns these exact sources. Do not introduce
# another ring / thread-local profiler instance through core_minimal.
get_target_property(core_sources foundation_core_minimal SOURCES)
list(FILTER core_sources EXCLUDE REGEX ".*/utils/(ProfilerRing|LiteTrace)\\.cpp$")
set_property(TARGET foundation_core_minimal PROPERTY SOURCES "${core_sources}")
target_link_libraries(foundation_core_minimal PUBLIC spatial::foundation_profiler_ring)
add_library(mng_host_jobs STATIC keyed_jobs.cpp)
target_link_libraries(mng_host_jobs PUBLIC spatial::foundation_async_task spatial::foundation_profiler_ring)
target_include_directories(mng_host_jobs PUBLIC .)
add_executable(mng_keyed_jobs_probe keyed_jobs_test.cpp)
target_link_libraries(mng_keyed_jobs_probe PRIVATE mng_host_jobs)
target_compile_options(mng_keyed_jobs_probe PRIVATE -UNDEBUG)
